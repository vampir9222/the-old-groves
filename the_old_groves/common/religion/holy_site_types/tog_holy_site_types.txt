# The Old Groves - holy sites of Druidism
# Three eminent sites, one per main branch, and two regular ones spread across the Celtic world.
# Iona is a vanilla holy site, shared with the Christians (see tog_faith_types.txt).

# Tara: seat of the High King of Ireland, the Lia Fail. Eminent (Gaelic).
# The hill of Tara lies between Trim and Navan: county of Dublin, barony of Trim.
# (c_tara is a Siberian county, not the Irish Tara.) Norse-held in 867.
tog_tara = {
	county = c_dublin
	barony = b_trim

	county_holder_character_modifier = {
		monthly_prestige_gain_mult = 0.1
	}

	county_modifier_scaled = {
		scale = holy_site_total_artifact_rarity
		development_growth_factor = 0.01
		county_opinion_add = 1
	}

	faith_character_modifier = {
		monthly_prestige_gain_mult = 0.05
		vassal_opinion = 3
	}

	faith_character_modifier_scaled = {
		scale = holy_site_total_artifact_rarity
		monthly_prestige_gain_mult = 0.01
	}
}
