# The Old Groves - names of the rites the game creates for Druidism (after the reform:
# unreformed faiths never spawn rites). The game takes the first unused name whose
# trigger is met, so this file must load before vanilla 00_rite_names.txt, whose
# generic names accept any founder. Order: Gaelic, Brythonic (Bretons too, even in
# Armorica), Gaulish (capital in Gaul), then the common list for anyone else.
# Loc: <name>, _adj, _adherent, _adherent_plural, _desc (one desc per branch).

tog_rite_name_filidecht = {
	name = tog_rite_filidecht
	trigger = {
		scope:faith.religion = religion:tog_celtic_religion
		scope:founder ?= { culture = { has_cultural_pillar = heritage_goidelic } }
	}
}
