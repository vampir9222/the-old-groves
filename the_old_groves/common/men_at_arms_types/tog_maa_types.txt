# The Old Groves - men-at-arms

# Scripted variables are local to each file: copied from vanilla 01_fp1_maa_types.txt
@cultural_maa_extra_ai_score = 60
@provisions_cost_infantry_cheap = 3

# Fénnidi: the hunter-warriors of a fianna, living in the wilds outside settled society.
# Troops of the Celtic military orders (holy_order_maa), also recruitable once the faith is reformed.
# The order's own leader always has them, so the Gosgordd keeps them before the reform.
# Weaker than the Jomsvikings, stronger than a cultural skirmisher.
tog_fennidi = {
	type = skirmishers

	can_recruit = {
		religion = religion:tog_celtic_religion
		OR = {
			faith = { NOT = { has_doctrine_parameter = unreformed } }
			government_has_flag = government_is_military_holy_order
		}
	}

	damage = 22
	toughness = 20
	pursuit = 25
	screen = 25

	terrain_bonus = {
		forest = { damage = 8 toughness = 4 }
		hills = { damage = 6 toughness = 4 }
		wetlands = { toughness = 4 }
	}

	counters = {
		heavy_infantry = 1
	}

	buy_cost = { gold = tog_fennidi_recruitment_cost }
	low_maintenance_cost = { gold = tog_fennidi_low_maint_cost }
	high_maintenance_cost = { gold = tog_fennidi_high_maint_cost }
	provision_cost = @provisions_cost_infantry_cheap

	stack = 100
	ai_quality = { value = @cultural_maa_extra_ai_score }
	icon = skirmishers
}
