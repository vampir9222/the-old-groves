# The Old Groves - artefact templates
# The artefacts of the quest are holy relics of Druidism: they can be enshrined in
# its holy sites, which activates them. Their bonuses work for anyone (the quest
# gives them before the founding).

	can_equip = {
		always = yes
	}

	can_benefit = {
		always = yes
	}

	is_holy_relic = {
		religion = religion:tog_celtic_religion
	}

	fallback = {
	}

	unique = yes
}
