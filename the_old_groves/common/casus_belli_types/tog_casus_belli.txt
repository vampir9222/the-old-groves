# The Old Groves - casus belli

# After the quest, the founder's dynasty can drive the foreigners (Saxons,
# Norsemen...) out of Britain, one duchy at a time. Modelled on fp2_expel_interloper.
# Landed rulers only: adventurers have tog_expel_saxons_adventurer_cb.
tog_expel_saxons_cb = {
	group = conquest
	icon = struggle

	combine_into_one = yes
	should_show_war_goal_subview = yes
	allow_hostages = no
	mutually_exclusive_titles = { always = yes }

	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	allowed_for_character = {
		religion = religion:tog_celtic_religion
		dynasty ?= { has_variable = tog_saxons_cb }
		NOT = { government_has_flag = government_is_landless_adventurer }
	}

	allowed_against_character = { tog_saxon_ruler_trigger = yes }

	target_titles = neighbor_land_or_water
	target_title_tier = duchy
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder

	valid_to_start = {
		scope:target = {
			any_de_jure_county = {
				title_province = { geographical_region = world_europe_west_britannia }
				holder.top_liege = scope:defender
			}
		}
	}

	on_invalidated_desc = msg_county_conquest_war_invalidated_message
	on_invalidated = {}

	cost = {
		piety = {
			value = medium_piety_value
			multiply = common_cb_piety_cost_multiplier
		}
		prestige = {
			value = minor_prestige_value
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		desc = tog_expel_saxons_victory_desc
	}

	on_victory = {
		add_legitimacy_attacker_victory_effect = yes
		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }

		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = title_change
			add_claim_on_loss = yes
		}
		every_in_list = {
			list = target_titles
			every_de_jure_county = {
				limit = {
					holder.top_liege = scope:defender
					NOR = {
						holder = scope:attacker
						holder.liege = scope:attacker
					}
				}
				change_title_holder = {
					holder = scope:attacker
					change = scope:title_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:title_change

		every_in_list = {
			list = target_titles
			scope:attacker = {
				add_prestige_experience = medium_prestige_value
				add_piety_experience = medium_piety_value
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = yes
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		desc = tog_expel_saxons_white_peace_desc
	}

	on_white_peace = {
		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }
		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
		}
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = yes
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		desc = tog_expel_saxons_defeat_desc
	}

	on_defeat = {
		add_legitimacy_attacker_defeat_effect = yes
		scope:defender = {
			accolade_defender_war_end_glory_gain_high_effect = yes
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
			}
		}
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 3
			}
			add_prestige = {
				value = major_prestige_value
				multiply = -1.0
			}
		}
		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = yes
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		add_truce_attacker_defeat_effect = yes
		scope:attacker = { save_temporary_scope_as = loser }
		on_lost_aggression_war_discontent_loss = yes
	}

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "TOG_EXPEL_SAXONS_WAR_NAME"
	war_name_base = "TOG_EXPEL_SAXONS_WAR_NAME_BASE"
	cb_name = "TOG_EXPEL_SAXONS_CB_NAME"

	interface_priority = 80

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 0.8

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}

# The same for an adventurer of the founder's dynasty: one county of Britain,
# which makes him a landed ruler. Modelled on ep3_laamp_adventurer_duchy_invasion.
tog_expel_saxons_adventurer_cb = {
	icon = invasion
	group = invasion

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }
	allow_hostages = no

	allowed_for_character = {
		religion = religion:tog_celtic_religion
		dynasty ?= { has_variable = tog_saxons_cb }
		ep3_adventurer_invasion_allowed_for_character_trigger = yes
	}

	allowed_for_character_display_regardless = {
		is_at_war = no
	}

	allowed_against_character = {
		tog_saxon_ruler_trigger = yes
		ep3_adventurer_landless_allowed_against_character_trigger = yes
	}

	target_titles = all
	target_title_tier = county
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder

	valid_to_start = {
		scope:target = {
			title_province = { geographical_region = world_europe_west_britannia }
			holder.top_liege = scope:defender
		}
	}

	should_invalidate = { ep3_adventurer_landless_should_invalidate_trigger = yes }

	on_invalidated_desc = msg_county_conquest_war_invalidated_message
	on_invalidated = {}

	cost = {
		piety = {
			value = minor_piety_value
			multiply = common_cb_piety_cost_multiplier
		}
		prestige = {
			value = minor_prestige_value
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		desc = tog_expel_saxons_adventurer_victory_desc
	}

	on_victory = {
		scope:attacker = {
			add_character_flag = {
				flag = domicile_culture_and_faith_check
				days = 1
			}
			ordered_in_list = {
				list = target_titles
				order_by = tier
				save_scope_as = target
			}
			ep3_become_landed_warning_effect = {
				TITLE_RECEIVER = scope:attacker
				TITLE = scope:target
			}
			ep3_landless_invasion_titles_taken_effect = {
				TITLE_GIVER = scope:defender
				TITLE_RECEIVER = scope:attacker
				TITLE = scope:target
				TITLE_LIST = target_titles
			}
			scope:target = { save_temporary_scope_as = new_capital }
			set_realm_capital = scope:new_capital
			custom_tooltip = fp1_varangian_adventurer_conquest.summaries.control_rebate
			trigger_event = fp1_varangian_adventures.0001
			ep3_become_landed_cleanup_effect = yes
			add_prestige_experience = medium_prestige_value
			add_piety_experience = medium_piety_value
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = yes
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		desc = tog_expel_saxons_white_peace_desc
	}

	on_white_peace = {
		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }
		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
		}
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		desc = tog_expel_saxons_defeat_desc
	}

	on_defeat = {
		scope:defender = {
			accolade_defender_war_end_glory_gain_high_effect = yes
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
			}
		}
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 3
			}
			add_prestige = {
				value = major_prestige_value
				multiply = -1.0
			}
		}
		add_truce_attacker_defeat_effect = yes
		scope:attacker = { save_temporary_scope_as = loser }
		on_lost_aggression_war_discontent_loss = yes
	}

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "TOG_EXPEL_SAXONS_WAR_NAME"
	war_name_base = "TOG_EXPEL_SAXONS_WAR_NAME_BASE"
	cb_name = "TOG_EXPEL_SAXONS_CB_NAME"

	interface_priority = 80

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 0.8

	max_attacker_score_from_battles = 75
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
