# The Old Groves - Gosgordd of Mynyddog

# Root: the founder. Copy of vanilla forge_the_jomsvikings_scripted_effect, with a fixed
# name and arms for the order, and Edinburgh renamed Din Eidyn when it hosts it.
tog_found_gosgordd_effect = {
	save_scope_as = founder
	# A county of the Old North, Lothian first when it is valid.
	random_held_title = {
		limit = { tog_valid_county_for_gosgordd_trigger = yes }
		weight = {
			base = 1
			modifier = {
				add = 100
				this = title:c_lothian
			}
		}
		save_scope_as = designated_county
	}
	# An empty slot gets a castle; otherwise a secondary castle or city.
	if = {
		limit = {
			scope:designated_county = {
				any_county_province = { has_holding = no }
			}
		}
		scope:designated_county = {
			random_county_province = {
				limit = { has_holding = no }
				set_holding_type = castle_holding
				barony = { save_scope_as = leased_barony }
			}
		}
	}
	else = {
		scope:designated_county = {
			random_county_province = {
				limit = {
					is_county_capital = no
					OR = {
						has_holding_type = castle_holding
						has_holding_type = city_holding
					}
				}
				barony = { save_scope_as = leased_barony }
			}
		}
	}
	# A barony held by a vassal comes back to the founder first, as in the vanilla decision.
	if = {
		limit = { scope:leased_barony.holder != root }
		create_title_and_vassal_change = {
			type = leased_out
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:leased_barony = {
			change_title_holder_include_vassals = {
				holder = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
	create_character = {
		template = holy_order_leader_character
		location = scope:leased_barony.title_province
		save_scope_as = leader
	}
	scope:leader = { set_character_rite ?= root.rite }
	create_holy_order_neutral_effect = {
		LEADER = scope:leader
		CAPITAL = scope:leased_barony
		NEW_HO_SCOPE = new_holy_order
		FOUNDER = scope:founder
		ORDER_TYPE = military
	}
	scope:leader.primary_title = {
		set_title_name = tog_holy_order_gosgordd
		set_coa = tog_gosgordd
	}
	if = {
		limit = { scope:leased_barony = title:b_edinburgh }
		scope:leased_barony = { set_title_name = tog_b_din_eidyn }
	}
	# Walls, barracks and military camps, with an extra slot when none is free.
	scope:leased_barony.title_province = {
		if = {
			limit = {
				NOT = { has_building_or_higher = curtain_walls_01 }
			}
			if = {
				limit = { free_building_slots = 0 }
				add_province_modifier = extra_building_slot
			}
			add_building = curtain_walls_01
		}
		if = {
			limit = {
				NOT = { has_building_or_higher = barracks_01 }
			}
			if = {
				limit = { free_building_slots = 0 }
				add_province_modifier = extra_building_slot
			}
			add_building = barracks_01
		}
		if = {
			limit = {
				NOT = { has_building_or_higher = military_camps_01 }
			}
			if = {
				limit = { free_building_slots = 0 }
				add_province_modifier = extra_building_slot
			}
			add_building = military_camps_01
		}
	}
	# Once per game.
	set_global_variable = {
		name = tog_gosgordd_founded
		value = scope:leader.primary_title
	}
	create_holy_order_accompanying_effect = yes
}
