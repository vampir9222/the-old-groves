# The Old Groves - restoration quest effects

# Starts the quest for the current character.
tog_start_quest_effect = {
	create_story = tog_story_old_groves
}

# Moves the quest to a new stage: STAGE and DESTINATION (a barony).
tog_next_stage_effect = {
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		set_variable = {
			name = tog_stage
			value = $STAGE$
		}
		set_variable = {
			name = tog_destination
			value = title:$DESTINATION$
		}
	}
}

# Sets out for the current quest destination, one way.
# Adventurers move their camp; rulers travel and stay until the next leg or the way home.
tog_travel_to_destination_effect = {
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		var:tog_destination ?= {
			title_province = { save_scope_as = tog_travel_target }
		}
	}
	if = {
		limit = {
			is_travelling = no
			exists = scope:tog_travel_target
			NOT = { location = scope:tog_travel_target }
		}
		if = {
			limit = { is_landless_adventurer = yes }
			start_travel_plan = {
				destination = scope:tog_travel_target
				on_arrival_event = tog_quest.0060
				on_arrival_destinations = last
				return_trip = no
				travel_with_domicile = yes
			}
		}
		else = {
			start_travel_plan = {
				destination = scope:tog_travel_target
				on_arrival_event = tog_quest.0060
				on_arrival_destinations = last
				return_trip = no
			}
		}
	}
}

# Rulers only: the way home at the end of the step.
tog_travel_home_effect = {
	if = {
		limit = {
			is_landless_adventurer = no
			is_travelling = no
			exists = capital_province
			NOT = { location = capital_province }
		}
		start_travel_plan = {
			destination = capital_province
			return_trip = no
		}
	}
}

# Bandits' camp: attacking (martial). Success frees the old man and yields the loot;
# failure wounds you and you come back a few days later.
tog_camp_attack_effect = {
	random_list = {
		0 = {
			modifier = { add = tog_good_weight_martial_value }
			custom_tooltip = tog_quest.0023.a.good
			add_gold = medium_gold_value
			add_martial_lifestyle_xp = minor_lifestyle_xp
			add_prestige = minor_prestige_gain
			hidden_effect = {
				trigger_event = {
					id = tog_quest.0050
					days = 1
				}
			}
		}
		60 = {
			custom_tooltip = tog_quest.0023.a.bad
			increase_wounds_no_death_effect = { REASON = fight }
			add_prestige = minor_prestige_loss
			hidden_effect = {
				trigger_event = {
					id = tog_quest.0024
					days = 1
				}
			}
		}
	}
}

# Bandits' camp: paying the ransom (certain).
tog_camp_pay_effect = {
	remove_short_term_gold = medium_gold_value
	custom_tooltip = tog_quest.0023.c.tt
	hidden_effect = {
		trigger_event = {
			id = tog_quest.0050
			days = 1
		}
	}
}

# Bandits' camp: leaving, to come back once there is enough gold or martial skill.
# Rulers head home; the camp can be faced again through the travel decision.
tog_camp_leave_effect = {
	custom_tooltip = tog_quest.0025.c.tt
	hidden_effect = {
		random_owned_story = {
			limit = { story_type = tog_story_old_groves }
			set_variable = {
				name = tog_left_camp
				value = yes
			}
		}
		tog_travel_home_effect = yes
	}
}

		}
		add_to_variable_list = {

			target = title:b_st_davids
		}
		add_to_variable_list = {
			name = tog_monasteries
			target = title:b_dunkeld
		}
		add_to_variable_list = {
			name = tog_monasteries
			target = title:b_applecross
		}
	}
}

# Sets out for a monastery (province PROV), one way.
tog_travel_to_province_effect = {
	if = {
		limit = {
			is_travelling = no
			NOT = { location = province:$PROV$ }
		}
		if = {
			limit = { is_landless_adventurer = yes }
			start_travel_plan = {
				destination = province:$PROV$
				on_arrival_event = tog_quest.0060
				on_arrival_destinations = last
				return_trip = no
				travel_with_domicile = yes
			}
		}
		else = {
			start_travel_plan = {
				destination = province:$PROV$
				on_arrival_event = tog_quest.0060
				on_arrival_destinations = last
				return_trip = no
			}
		}
	}
	# Road events: the roll checks that the journey is under way. Only one roll
	# can be pending at a time, so several trips never stack their rolls.
	hidden_effect = {
		if = {
			limit = { NOT = { exists = var:tog_tev_pending } }
			set_variable = { name = tog_tev_pending value = yes }
			trigger_event = {
				id = tog_quest.0150
				days = { 25 38 }
			}
		}
	}
}

# The book of monastery M is obtained: remove it from the list, read it in 2 to 4 months.
tog_monastery_book_effect = {
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		set_variable = {
			name = tog_m_$M$_done
			value = yes
		}
		if = {
			limit = { NOT = { exists = var:tog_books_got } }
			set_variable = {
				name = tog_books_got
				value = 0
			}
		}
		change_variable = {
			name = tog_books_got
			add = 1
		}
		set_variable = {
			name = tog_m_$M$_toread
			value = var:tog_books_got
		}
		remove_list_variable = {
			name = tog_monasteries
			target = title:$TITLE$
		}
	}
	custom_tooltip = tog_m.book_tt
	hidden_effect = { tog_next_reading_effect = yes }
}

# Closes a method at monastery M (FLAG = talk, pay or steal) and shows the failure scene FAIL.
tog_monastery_close_effect = {
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		set_variable = {
			name = tog_m_$M$_no_$FLAG$
			value = yes
		}
		set_variable = {
			name = tog_current_m
			value = flag:$M$
		}
	}
	hidden_effect = {
		trigger_event = {
			id = $FAIL$
			days = 1
		}
	}
}

# Paying a monk: fixed odds, gold only. A refusal costs nothing; a monk who keeps

				M = $M$

				FAIL = tog_quest.0163

		}
	}
}

# Forcing the doors (martial). Every attack puts the other monasteries on their guard.
# Success: the book and some loot, but piety lost and the local lord is angry.
tog_monastery_force_effect = {
	random_list = {
		100 = {
			custom_tooltip = tog_m.force_good
			hidden_effect = { trigger_event = tog_quest.0174 }
			add_gold = minor_gold_value
			add_prestige = medium_prestige_gain
			add_piety = medium_piety_loss
			province:$PROV$.county.holder ?= {
				if = {
					limit = { NOT = { this = root } }
					add_opinion = {
						target = root
						modifier = angry_opinion
						opinion = -30
					}
				}
			}
			tog_monastery_book_effect = {
				M = $M$
				TITLE = $TITLE$
			}
		}
		0 = {
			modifier = { add = tog_force_fail_value }
			custom_tooltip = tog_m.force_bad
			add_prestige = minor_prestige_loss
			hidden_effect = {
				random_owned_story = {
					limit = { story_type = tog_story_old_groves }
					set_variable = {
						name = tog_current_m
						value = flag:$M$
					}
				}
				trigger_event = {
					id = tog_quest.0165
					days = 1
				}
			}
		}
	}
	hidden_effect = {
		if = {
			limit = { NOT = { exists = var:tog_raids } }
			set_variable = {
				name = tog_raids
				value = 0
			}
		}
		change_variable = {
			name = tog_raids
			add = 1
		}
	}
}

# A book has been read; after the fifth, the synthesis.
tog_book_read_effect = {
	add_learning_lifestyle_xp = medium_lifestyle_xp
	hidden_effect = {
		random_owned_story = {
			limit = { story_type = tog_story_old_groves }
			if = {
				limit = { NOT = { exists = var:tog_books_read } }
				set_variable = {
					name = tog_books_read
					value = 0
				}
			}
			change_variable = {
				name = tog_books_read
				add = 1
			}
			remove_variable = tog_reading
			if = {
				limit = { var:tog_books_read >= 5 }
				story_owner = {
					trigger_event = {
						id = tog_quest.0130
						days = 1
					}
				}
			}
		}
		tog_next_reading_effect = yes
	}
}

# After a failure scene: back to the arrival scene of the current monastery.
tog_reopen_monastery_effect = {
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		save_scope_as = tog_story
	}
	if = {
		limit = { scope:tog_story.var:tog_current_m ?= flag:jarrow }
		trigger_event = tog_quest.0101
	}
	else_if = {
		limit = { scope:tog_story.var:tog_current_m ?= flag:whithorn }
		trigger_event = tog_quest.0102
	}
	else_if = {
		limit = { scope:tog_story.var:tog_current_m ?= flag:mynyw }
		trigger_event = tog_quest.0103
	}
	else_if = {
		limit = { scope:tog_story.var:tog_current_m ?= flag:dunkeld }
		trigger_event = tog_quest.0104
	}
	else_if = {
		limit = { scope:tog_story.var:tog_current_m ?= flag:applecross }
		trigger_event = tog_quest.0105
	}
}


# Saves the companions as scope:tog_peredur and scope:tog_lailoken, for event portraits.

# Books are read one at a time, in the order they were obtained: starts the next
# waiting book (number tog_books_read + 1), if no reading is under way.
tog_next_reading_effect = {
	random_owned_story = {
		limit = {
			story_type = tog_story_old_groves
			NOT = { exists = var:tog_reading }
		}
		save_scope_as = tog_story
	}
	scope:tog_story ?= {
		if = {
			limit = { var:tog_m_jarrow_toread ?= tog_next_book_value }
			remove_variable = tog_m_jarrow_toread
			set_variable = {
				name = tog_reading
				value = yes
			}
			story_owner = {
				trigger_event = {
					id = tog_quest.0121
					days = { 60 120 }
				}
			}
		}
		else_if = {
			limit = { var:tog_m_whithorn_toread ?= tog_next_book_value }
			remove_variable = tog_m_whithorn_toread
			set_variable = {
				name = tog_reading
				value = yes
			}
			story_owner = {
				trigger_event = {
					id = tog_quest.0122
					days = { 60 120 }
				}
			}
		}
		else_if = {
			limit = { var:tog_m_mynyw_toread ?= tog_next_book_value }
			remove_variable = tog_m_mynyw_toread
			set_variable = {
				name = tog_reading
				value = yes
			}
			story_owner = {
				trigger_event = {
					id = tog_quest.0123
					days = { 60 120 }
				}
			}
		}
		else_if = {
			limit = { var:tog_m_dunkeld_toread ?= tog_next_book_value }
			remove_variable = tog_m_dunkeld_toread
			set_variable = {
				name = tog_reading
				value = yes
			}
			story_owner = {
				trigger_event = {
					id = tog_quest.0124
					days = { 60 120 }
				}
			}
		}
		else_if = {
			limit = { var:tog_m_applecross_toread ?= tog_next_book_value }
			remove_variable = tog_m_applecross_toread
			set_variable = {
				name = tog_reading
				value = yes
			}
			story_owner = {
				trigger_event = {
					id = tog_quest.0125
					days = { 60 120 }
				}
			}
		}
	}
}
