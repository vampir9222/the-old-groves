# The Old Groves - restoration quest effects

# Starts the quest for the current character.
tog_start_quest_effect = {
	create_story = tog_story_old_groves
}

# Moves the quest to a new stage: STAGE and DESTINATION (a barony).
tog_next_stage_effect = {
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		set_variable = {
			name = tog_stage
			value = $STAGE$
		}
		set_variable = {
			name = tog_destination
			value = title:$DESTINATION$
		}
	}
}

# Sets out for the current quest destination, one way.
# Adventurers move their camp; rulers travel and stay until the next leg or the way home.
tog_travel_to_destination_effect = {
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		var:tog_destination ?= {
			title_province = { save_scope_as = tog_travel_target }
		}
	}
	if = {
		limit = {
			is_travelling = no
			exists = scope:tog_travel_target
			NOT = { location = scope:tog_travel_target }
		}
		if = {
			limit = { is_landless_adventurer = yes }
			start_travel_plan = {
				destination = scope:tog_travel_target
				on_arrival_event = tog_quest.0060
				on_arrival_destinations = last
				return_trip = no
				travel_with_domicile = yes
			}
		}
		else = {
			start_travel_plan = {
				destination = scope:tog_travel_target
				on_arrival_event = tog_quest.0060
				on_arrival_destinations = last
				return_trip = no
			}
		}
	}
	# Road events (the roll only fires from step 2 on, while travelling).
	hidden_effect = {
		if = {
			limit = { NOT = { exists = var:tog_tev_pending } }
			set_variable = { name = tog_tev_pending value = yes }
			trigger_event = {
				id = tog_quest.0150
				days = { 45 75 }
			}
		}
	}
}

# Rulers only: the way home at the end of the step.
tog_travel_home_effect = {
	if = {
		limit = {
			is_landless_adventurer = no
			is_travelling = no
			exists = capital_province
			NOT = { location = capital_province }
		}
		start_travel_plan = {
			destination = capital_province
			return_trip = no
		}
	}
}

# Bandits' camp: attacking (martial). Success frees the old man and yields the loot,
# though you may be wounded; failure wounds you and you come back a few days later.

				chance = tog_wound_chance_value

			}
			hidden_effect = {
				trigger_event = {
					id = tog_quest.0050
					days = 1
				}
			}
		}
		60 = {
			custom_tooltip = tog_quest.0023.a.bad
			increase_wounds_no_death_effect = { REASON = fight }

				trigger_event = {
					id = tog_quest.0024
					days = 1
				}
			}

	}
}

# Sets out for a monastery (province PROV), one way.
tog_travel_to_province_effect = {
	if = {
		limit = {
			is_travelling = no
			NOT = { location = province:$PROV$ }
		}
		if = {
			limit = { is_landless_adventurer = yes }
			start_travel_plan = {
				destination = province:$PROV$
				on_arrival_event = tog_quest.0060
				on_arrival_destinations = last
				return_trip = no
				travel_with_domicile = yes
			}
		}
		else = {
			start_travel_plan = {
				destination = province:$PROV$
				on_arrival_event = tog_quest.0060
				on_arrival_destinations = last
				return_trip = no
			}
		}
	}
	# Road events (one every six months of travel or so): the roll checks that the journey is under way. Only one roll
	# can be pending at a time, so several trips never stack their rolls.
	hidden_effect = {
		if = {
			limit = { NOT = { exists = var:tog_tev_pending } }
			set_variable = { name = tog_tev_pending value = yes }
			trigger_event = {
				id = tog_quest.0150
				days = { 45 75 }
			}
		}
	}
}

# The book of monastery M is obtained: remove it from the list, read it in 2 to 4 months.
tog_monastery_book_effect = {
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		set_variable = {
			name = tog_m_$M$_done
			value = yes
		}
		if = {
			limit = { NOT = { exists = var:tog_books_got } }
			set_variable = {
				name = tog_books_got
				value = 0
			}
		}
		change_variable = {
			name = tog_books_got
			add = 1
		}
		set_variable = {
			name = tog_m_$M$_toread
			value = var:tog_books_got
		}
		remove_list_variable = {
			name = tog_monasteries
			target = title:$TITLE$
		}
	}
	custom_tooltip = tog_m.book_tt
	hidden_effect = { tog_next_reading_effect = yes }
}

# Closes a method at monastery M (FLAG = talk, pay or steal) and shows the failure scene FAIL.
tog_monastery_close_effect = {
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		set_variable = {
			name = tog_m_$M$_no_$FLAG$
			value = yes
		}
		set_variable = {
			name = tog_current_m
			value = flag:$M$
		}
	}
	hidden_effect = {
		trigger_event = {
			id = $FAIL$
			days = 1
		}
	}
}

# Forcing the doors (martial). Every attack puts the other monasteries on their guard.
# Attacking monks costs piety, whatever the outcome. Success: the book, some loot and
# prestige, but the local lord is angry. Either way, you may be wounded (the lower
# your martial, the likelier).
tog_monastery_force_effect = {
	add_piety = medium_piety_loss
	random = {
		chance = tog_wound_chance_value
		increase_wounds_no_death_effect = { REASON = fight }
	}
	random_list = {
		100 = {
			custom_tooltip = tog_m.force_good
			hidden_effect = { trigger_event = tog_quest.0174 }
			add_gold = minor_gold_value
			add_prestige = medium_prestige_gain
			province:$PROV$.county.holder ?= {
				if = {
					limit = { NOT = { this = root } }
					add_opinion = {
						target = root
						modifier = angry_opinion
						opinion = -30
					}
				}
			}
			tog_monastery_book_effect = {
				M = $M$
				TITLE = $TITLE$
			}
		}
		0 = {
			modifier = { add = tog_force_fail_value }
			custom_tooltip = tog_m.force_bad
			add_prestige = minor_prestige_loss
			hidden_effect = {
				random_owned_story = {
					limit = { story_type = tog_story_old_groves }
					set_variable = {
						name = tog_current_m
						value = flag:$M$
					}
				}
				trigger_event = {
					id = tog_quest.0165
					days = 1
				}
			}
		}
	}
	hidden_effect = {
		if = {
			limit = { NOT = { exists = var:tog_raids } }
			set_variable = {
				name = tog_raids
				value = 0
			}
		}
		change_variable = {
			name = tog_raids
			add = 1
		}
	}
}

# A book has been read; after the fifth, the synthesis.
tog_book_read_effect = {
	add_learning_lifestyle_xp = medium_lifestyle_xp
	hidden_effect = {
		random_owned_story = {
			limit = { story_type = tog_story_old_groves }
			if = {
				limit = { NOT = { exists = var:tog_books_read } }
				set_variable = {
					name = tog_books_read
					value = 0
				}
			}
			change_variable = {
				name = tog_books_read
				add = 1
			}
			remove_variable = tog_reading
			if = {
				limit = { var:tog_books_read >= 5 }
				story_owner = {
					trigger_event = {
						id = tog_quest.0130
						days = 1
					}
				}
			}
		}
		tog_next_reading_effect = yes
	}
}

# After a failure scene: back to the arrival scene of the current monastery.
tog_reopen_monastery_effect = {
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		save_scope_as = tog_story
	}
	if = {
		limit = { scope:tog_story.var:tog_current_m ?= flag:jarrow }
		trigger_event = tog_quest.0101
	}
	else_if = {
		limit = { scope:tog_story.var:tog_current_m ?= flag:whithorn }
		trigger_event = tog_quest.0102
	}
	else_if = {
		limit = { scope:tog_story.var:tog_current_m ?= flag:mynyw }
		trigger_event = tog_quest.0103
	}
	else_if = {
		limit = { scope:tog_story.var:tog_current_m ?= flag:dunkeld }
		trigger_event = tog_quest.0104
	}
	else_if = {
		limit = { scope:tog_story.var:tog_current_m ?= flag:applecross }
		trigger_event = tog_quest.0105
	}
}

# Books are read one at a time, in the order they were obtained: starts the next
# waiting book (number tog_books_read + 1), if no reading is under way.
tog_next_reading_effect = {
	random_owned_story = {
		limit = {
			story_type = tog_story_old_groves
			NOT = { exists = var:tog_reading }
		}
		save_scope_as = tog_story
	}
	scope:tog_story ?= {
		if = {
			limit = { var:tog_m_jarrow_toread ?= tog_next_book_value }
			remove_variable = tog_m_jarrow_toread
			set_variable = {
				name = tog_reading
				value = yes
			}
			story_owner = {
				trigger_event = {
					id = tog_quest.0121
					days = { 60 120 }
				}
			}
		}
		else_if = {
			limit = { var:tog_m_whithorn_toread ?= tog_next_book_value }
			remove_variable = tog_m_whithorn_toread
			set_variable = {
				name = tog_reading
				value = yes
			}
			story_owner = {
				trigger_event = {
					id = tog_quest.0122
					days = { 60 120 }
				}
			}
		}
		else_if = {
			limit = { var:tog_m_mynyw_toread ?= tog_next_book_value }
			remove_variable = tog_m_mynyw_toread
			set_variable = {
				name = tog_reading
				value = yes
			}
			story_owner = {
				trigger_event = {
					id = tog_quest.0123
					days = { 60 120 }

			}
		}
		else_if = {
			limit = { var:tog_m_dunkeld_toread ?= tog_next_book_value }
			remove_variable = tog_m_dunkeld_toread

# Place P (bru, uisneach or mabon, title TITLE) is done.

		}
		remove_list_variable = {
			name = tog_places

			}

				trigger_event = {
					id = tog_quest.0220
					days = 3
				}
			}
		}
	}
}

# Companion scope:WHO joins the court and the quest, saved in the story as var WHO.
tog_join_quest_effect = {
	add_courtier = scope:$WHO$
	# The family follows: spouse and children who hold no land.
	scope:$WHO$ = {
		every_spouse = {
			limit = { is_ruler = no }
			save_temporary_scope_as = tog_kin
			root = { add_courtier = scope:tog_kin }
		}
		every_child = {
			limit = { is_ruler = no }
			save_temporary_scope_as = tog_kin
			root = { add_courtier = scope:tog_kin }
		}
	}
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		set_variable = {
			name = $WHO$
			value = scope:$WHO$
		}
		add_to_variable_list = {
			name = tog_companions
			target = scope:$WHO$
		}
	}
}

# when created (quest or pilgrimage), and is never created twice.

	if = {
		limit = { NOT = { exists = global_var:tog_relic_sickle } }
		set_global_variable = { name = tog_relic_sickle value = yes }
		save_scope_as = owner # the artifact visuals read scope:owner
		create_artifact = {
			name = tog_golden_sickle_name
			description = tog_golden_sickle_description
			template = tog_relic_template
			type = pedestal

# Sends companion WHO (story variable) to PROV to win over a count there.
# The count is the county's holder if he is a count, else one of his count
# vassals; never the owner nor his liege, never one of the faith already.
tog_send_companion_effect = {
	random_owned_story = {
		limit = { story_type = tog_story_old_groves }
		var:$WHO$ ?= {
			if = {
				limit = {
					is_alive = yes
					is_courtier_of = root
					is_travelling = no
				}
				save_scope_as = tog_missionary
			}
		}
	}
	if = {
		limit = { exists = scope:tog_missionary }
		province:$PROV$.county.holder ?= {
			if = {
				limit = {
					highest_held_title_tier = tier_county
					tog_mission_target_trigger = yes
				}
				save_scope_as = tog_mission_target
			}
			else = {
				random_vassal = {
					limit = {
						highest_held_title_tier = tier_county
						tog_mission_target_trigger = yes
					}
					save_scope_as = tog_mission_target
				}
			}
		}
		if = {
			limit = { exists = scope:tog_mission_target }
			scope:tog_missionary = {
				set_variable = {
					name = tog_mission_target
					value = scope:tog_mission_target
				}
				set_variable = {
					name = tog_mission_owner
					value = root
				}
				start_travel_plan = {
					destination = province:$PROV$
					on_arrival_event = tog_quest.0711
					on_arrival_destinations = last
					return_trip = yes
				}
			}
			random_owned_story = {
				limit = { story_type = tog_story_old_groves }
				change_variable = {
					name = tog_mission_pending
					add = 1
				}
				# For the recap in the quest window.
				set_variable = {
					name = tog_res_$WHO$
					value = flag:pending
				}
			}
		}
	}
	clear_saved_scope = tog_missionary
	clear_saved_scope = tog_mission_target
}

# Approximate distance in km from the character to province PROV, kept in the
# character variable tog_km_NAME for the destination choices (0140, 0240).
# The game gives the squared map distance: its square root by Newton's method,
# then about 1.5 km per map pixel.
tog_distance_effect = {
	province:$PROV$ = { save_scope_as = tog_dist_target }
	set_local_variable = {
		name = tog_dist_sq
		value = "location.squared_distance(scope:tog_dist_target)"
	}
	set_local_variable = {
		name = tog_dist_x
		value = 300
	}
	while = {
		count = 8
		set_local_variable = {
			name = tog_dist_x
			value = {
				value = local_var:tog_dist_sq
				divide = local_var:tog_dist_x
				add = local_var:tog_dist_x
				divide = 2
			}
		}
	}
	set_variable = {
		name = tog_km_$NAME$
		value = {
			value = local_var:tog_dist_x
			multiply = 0.15
			round = yes
			multiply = 10
		}
	}
}

# Adds the tent to the owner's camp if a slot is free, and raises it to level 2
# after the miracle. The building cannot be built by hand: its can_construct asks
# for this short-lived variable, which only the quest sets.
tog_add_offerings_tent_effect = {
	set_variable = {
		name = tog_offerings_tent_gift
		value = yes
		days = 1
	}
	if = {
		limit = {
			any_owned_story = {
				story_type = tog_story_old_groves
				var:tog_stage >= 100
			}
		}
		save_temporary_scope_value_as = {
			name = tog_tent_level_2
			value = yes
		}
	}
	domicile ?= {
		if = {
			limit = {
				NOT = { has_domicile_building_or_higher = tog_offerings_tent }
				free_external_domicile_building_slots >= 1
			}
			add_domicile_building = tog_offerings_tent
		}
		if = {
			limit = {
				exists = scope:tog_tent_level_2
				has_domicile_building = tog_offerings_tent
			}
			add_domicile_building = tog_offerings_tent_02
		}
	}
}

# The stone of Mabon: one way of getting through, with odds from a skill (WEIGHT,
# against 60). Success leads to NEXT; failure closes this way (FLAG) and the
# scene (SELF) opens again for the other ways.
tog_mabon_try_effect = {
	random_list = {
		0 = {
			modifier = { add = $WEIGHT$ }
			custom_tooltip = $GOOD$
			add_prestige = minor_prestige_gain
			hidden_effect = { trigger_event = $NEXT$ }
		}
		# The failure weight falls as the skill rises (WEIGHT = 4 x skill + 20):
		# about 29% at 5, 43% at 10, 57% at 15, 71% at 20, 86% at 25.
		0 = {
			modifier = {
				add = {
					value = 140
					subtract = $WEIGHT$
					min = 5
				}
			}
			custom_tooltip = $BAD$
			hidden_effect = {
				random_owned_story = {
					limit = { story_type = tog_story_old_groves }
					set_variable = tog_mabon_no_$FLAG$
				}
				trigger_event = {
					id = $SELF$
					days = 1
				}
			}
		}
	}
}
