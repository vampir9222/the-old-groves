# The Old Groves - restoration quest values

# Weight of the good outcome in quest events, against a fixed 60 for the bad one.
# Skill 10 gives an even chance; skill 20, about 62 %; skill 5, about 40 %.
tog_good_weight_learning_value = {
	value = learning
	multiply = 4
	add = 20
}
tog_good_weight_diplomacy_value = {
	value = diplomacy
	multiply = 4
	add = 20
}
tog_good_weight_martial_value = {
	value = martial
	multiply = 4
	add = 20
}
tog_good_weight_intrigue_value = {
	value = intrigue
	multiply = 4
	add = 20
}

# Forcing a monastery's doors: martial, minus 15 for every monastery already
# attacked (the others are on their guard).
# Failure weight when forcing the doors, against 100 for success. Every attack puts
# the other monasteries on their guard: 0% failure, then about 30%, 60%, 85%.
tog_force_fail_value = {
	value = 0
	if = {
		limit = { var:tog_raids ?= 1 }
		add = 40
	}
	else_if = {
		limit = { var:tog_raids ?= 2 }
		add = 150
	}
	else_if = {
		limit = {
			exists = var:tog_raids
			var:tog_raids >= 3
		}
		add = 600
	}
}
tog_good_weight_prowess_value = {
	value = prowess
	multiply = 4
	add = 20
}

# Story scope: the number of the next book to read, in the order the books were obtained.
tog_next_book_value = {
	value = 0
	if = {
		limit = { exists = var:tog_books_read }
		add = var:tog_books_read
	}
	add = 1
}
