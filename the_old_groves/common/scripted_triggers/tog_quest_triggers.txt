# The Old Groves - restoration quest triggers

# Step 2: is a method still open at monastery M? FLAG = talk, pay or steal.
tog_monastery_open_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		NOT = { exists = var:tog_m_$M$_no_$FLAG$ }
	}
}

# Step 2: has the book of monastery M not been obtained yet?
tog_monastery_todo_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		var:tog_stage = 10
		NOT = { exists = var:tog_m_$M$_done }
	}
}

# Lailoken has joined the quest and is still alive.
tog_lailoken_alive_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		var:tog_lailoken ?= { is_alive = yes }
	}
}

# Peredur has joined the quest and is still alive.
tog_peredur_alive_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		var:tog_peredur ?= { is_alive = yes }
	}
}

# Step 3: has sacred place P (bru, uisneach or mabon) not been done yet?
tog_place_todo_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		var:tog_stage = 20
		NOT = { exists = var:tog_p_$P$_done }
	}
}

# Step 3: the stone of Mabon comes last, after Brú na Bóinne and Uisneach.
tog_mabon_open_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		exists = var:tog_p_bru_done
		exists = var:tog_p_uisneach_done
	}
}

# Companion WHO (story variable: tog_eogan, tog_gwion, tog_gofraid...) has joined
# the quest and is still alive.
tog_companion_alive_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		var:$WHO$ ?= { is_alive = yes }
	}
}

# The quest is not waiting for a minor heir to come of age.
tog_quest_not_paused_trigger = {
	NOT = {
		any_owned_story = {
			story_type = tog_story_old_groves
			exists = var:tog_paused
		}
	}
}
