# The Old Groves - restoration quest triggers

# Step 2: is a method still open at monastery M? FLAG = talk, pay or steal.
tog_monastery_open_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		NOT = { exists = var:tog_m_$M$_no_$FLAG$ }
	}
}

# Step 2: has the book of monastery M not been obtained yet?
tog_monastery_todo_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		var:tog_stage = 10
		NOT = { exists = var:tog_m_$M$_done }
	}
}

# Lailoken (the old man himself, not an heir) has joined the quest and is still alive.
tog_lailoken_alive_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		var:tog_lailoken ?= { is_alive = yes NOT = { has_character_flag = tog_heir_companion } }
	}
}

# Peredur has joined the quest and is still alive.
tog_peredur_alive_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		var:tog_peredur ?= { is_alive = yes }
	}
}

# Step 3: has sacred place P (bru, uisneach or mabon) not been done yet?
tog_place_todo_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		var:tog_stage = 20
		NOT = { exists = var:tog_p_$P$_done }
	}
}

# Step 3: the stone of Mabon comes last, after Brú na Bóinne and Uisneach.
tog_mabon_open_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		exists = var:tog_p_bru_done
		exists = var:tog_p_uisneach_done
	}
}

# Companion WHO (story variable: tog_eogan, tog_gwion, tog_gofraid...) has joined
# the quest and is still alive.
tog_companion_alive_trigger = {
	any_owned_story = {
		story_type = tog_story_old_groves
		var:$WHO$ ?= { is_alive = yes }
	}
}

# The quest is not waiting for a minor heir to come of age.
tog_quest_not_paused_trigger = {
	NOT = {
		any_owned_story = {
			story_type = tog_story_old_groves
			exists = var:tog_paused
		}
	}
}

# Step 8: a count a missionary may try to win over.
tog_mission_target_trigger = {
	is_alive = yes
	is_ai = yes
	NOT = { this = root }
	NOT = { root.liege ?= this }
	NOT = { faith = faith:tog_druidism }
}

# Event with scope:tog_missionary and scope:tog_story: the missionary is companion WHO.
tog_is_companion_trigger = {
	scope:tog_story.var:$WHO$ ?= scope:tog_missionary
}

# Story scope: Lailoken, with the Blood of Myrddin, is alive at the owner's court.
tog_second_sight_valid_trigger = {
	exists = var:tog_lailoken
	var:tog_lailoken = {
		is_alive = yes
		has_trait = tog_blood_of_myrddin
		is_courtier_of = root.story_owner
	}
}

# After the quest: a ruler holding land in Britain or Ireland who is a foreigner
# (neither Brythonic nor Goidelic) or not of the Celtic religion (a Christian
# Gael too), a target for "Drive out the Saxons".
tog_saxon_ruler_trigger = {
	OR = {
		NOT = { religion = religion:tog_celtic_religion }
		culture = {
			NOR = {
				has_cultural_pillar = heritage_goidelic
				has_cultural_pillar = heritage_brythonic
			}
		}
	}
	any_realm_county = {
		title_province = { geographical_region = world_europe_west_britannia }
	}
}
