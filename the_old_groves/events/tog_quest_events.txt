# The bandits' camp: attack (martial), talk (diplomacy) or pay (gold, certain).
# A failed attack wounds you; you come back a few days later and may attack again.
# A failed talk opens tog_quest.0025 at once: attack, pay, or leave and come back later.
tog_quest.0023 = {
	type = character_event
	title = tog_quest.0023.t
	desc = tog_quest.0023.desc
	theme = martial
	override_background = { reference = army_camp }
	left_portrait = {
		character = root
		animation = personality_bold
	}

	option = { # Attack the camp
		name = tog_quest.0023.a
		trigger = { martial >= 8 }
		show_as_unavailable = { always = yes }
		tog_camp_attack_effect = yes
	}

	option = { # Talk them round
		name = tog_quest.0023.b
		trigger = {
			any_owned_story = {
				story_type = tog_story_old_groves
				NOT = { exists = var:tog_tried_talk }
			}
		}
		random_list = {
			0 = {
				modifier = { add = tog_good_weight_diplomacy_value }
				custom_tooltip = tog_quest.0023.b.good
				add_diplomacy_lifestyle_xp = minor_lifestyle_xp
				hidden_effect = {
					trigger_event = {
						id = tog_quest.0050
						days = 1
					}
				}
			}
			60 = {
				custom_tooltip = tog_quest.0023.b.bad
				add_prestige = minor_prestige_loss
				hidden_effect = {
					random_owned_story = {
						limit = { story_type = tog_story_old_groves }
						set_variable = {
							name = tog_tried_talk
							value = yes
						}
					}
					trigger_event = tog_quest.0025
				}
			}
		}
	}

	option = { # Pay the ransom
		name = tog_quest.0023.c
		trigger = { gold >= medium_gold_value }
		show_as_unavailable = { always = yes }
		tog_camp_pay_effect = yes
	}

	option = { # Leave, and come back when ready
		name = tog_quest.0025.c
		tog_camp_leave_effect = yes
	}
}

# Arrival at a quest destination (travel, point of interest, or the monthly check).
tog_quest.0060 = {
	type = character_event
	hidden = yes

	immediate = {
		random_owned_story = {
			limit = { story_type = tog_story_old_groves }
			save_scope_as = tog_story
		}
		if = {
			limit = { exists = scope:tog_story }
			scope:tog_story = {
				if = {
					limit = {
						var:tog_stage = 1
						root.location = province:1692
						NOT = { exists = var:tog_seen_stone }
					}
					set_variable = {
						name = tog_seen_stone
						value = yes
					}
					root = { trigger_event = tog_quest.0003 }
				}
				else_if = {
					limit = {
						var:tog_stage = 2
						root.location = province:1704
						NOT = { exists = var:tog_v_wrong }
					}
					set_variable = {
						name = tog_v_wrong
						value = yes
					}
					root = { trigger_event = tog_quest.0021 }
				}
				else_if = {
					limit = {
						var:tog_stage = 3
						root.location = province:35
						NOT = { exists = var:tog_v_house }
					}
					set_variable = {
						name = tog_v_house
						value = yes
					}
					root = { trigger_event = tog_quest.0022 }
				}
				else_if = {
					limit = {
						var:tog_stage = 4
						root.location = province:1701
						OR = {
							NOT = { exists = var:tog_v_camp }
							exists = var:tog_returning
						}
					}
					set_variable = {
						name = tog_v_camp
						value = yes
					}
					remove_variable = tog_returning
					remove_variable = tog_left_camp
					root = { trigger_event = tog_quest.0023 }
				}
				else_if = {
					limit = {
						var:tog_stage = 10
						root.location = province:1606
						NOT = { exists = var:tog_m_jarrow_done }
					}
					root = { trigger_event = tog_quest.0101 }
				}
				else_if = {
					limit = {
						var:tog_stage = 10
						root.location = province:1684
						NOT = { exists = var:tog_m_whithorn_done }
					}
					root = { trigger_event = tog_quest.0102 }
				}
				else_if = {
					limit = {
						var:tog_stage = 10
						root.location = province:1655
						NOT = { exists = var:tog_m_mynyw_done }
					}
					root = { trigger_event = tog_quest.0103 }
				}
				else_if = {
					limit = {
						var:tog_stage = 10
						root.location = province:1743
						NOT = { exists = var:tog_m_dunkeld_done }
					}
					root = { trigger_event = tog_quest.0104 }
				}
				else_if = {
					limit = {
						var:tog_stage = 10
