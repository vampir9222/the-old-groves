# The bandits' camp: attack (martial), talk (diplomacy) or pay (gold, certain).
# A failed attack wounds you; you come back a few days later and may attack again.
# A failed talk opens tog_quest.0025 at once: attack, pay, or leave and come back later.
tog_quest.0023 = {
	type = character_event
	title = tog_quest.0023.t
	desc = tog_quest.0023.desc
	theme = martial
	override_background = { reference = army_camp }
	left_portrait = {
		character = root
		animation = personality_bold
	}

	option = { # Attack the camp
		name = tog_quest.0023.a
		trigger = { martial >= 8 }
		show_as_unavailable = { always = yes }
		tog_camp_attack_effect = yes
	}

	option = { # Talk them round
		name = tog_quest.0023.b
		trigger = {
			any_owned_story = {
				story_type = tog_story_old_groves
				NOT = { exists = var:tog_tried_talk }
			}
		}
		random_list = {
			0 = {
				modifier = { add = tog_good_weight_diplomacy_value }
				custom_tooltip = tog_quest.0023.b.good
				add_diplomacy_lifestyle_xp = minor_lifestyle_xp
				hidden_effect = {
					trigger_event = {
						id = tog_quest.0050
						days = 1
					}
				}
			}
			60 = {
				custom_tooltip = tog_quest.0023.b.bad
				add_prestige = minor_prestige_loss
				hidden_effect = {
					random_owned_story = {
						limit = { story_type = tog_story_old_groves }
						set_variable = {
							name = tog_tried_talk
							value = yes
						}
					}
					trigger_event = tog_quest.0025
				}
			}
		}
	}

	option = { # Pay the ransom
		name = tog_quest.0023.c
		trigger = { gold >= medium_gold_value }
		show_as_unavailable = { always = yes }
		tog_camp_pay_effect = yes
	}

	option = { # Leave, and come back when ready
		name = tog_quest.0025.c
		tog_camp_leave_effect = yes
	}
}

tog_quest.0711 = {
	type = character_event
	hidden = yes

	immediate = {
		save_scope_as = tog_missionary
		var:tog_mission_target ?= { save_scope_as = tog_mission_target }
		var:tog_mission_owner ?= { save_scope_as = tog_mission_owner }
		scope:tog_mission_owner ?= {
			random_owned_story = {
				limit = { story_type = tog_story_old_groves }
				save_scope_as = tog_story
			}
		}
		if = {
			limit = {
				exists = scope:tog_mission_target
				exists = scope:tog_story
				scope:tog_mission_target = { is_alive = yes }
			}
			# Five outcomes; the better the missionary thinks of the owner, the
			# better the odds.
			random_list = {
				# A triumph: the count converts and sends a gift.
				10 = {
					opinion_modifier = {
						who = scope:tog_missionary
						opinion_target = scope:tog_mission_owner
						multiplier = 0.3
					}
					save_scope_value_as = { name = tog_mission_result value = flag:triumph }
					tog_mission_convert_effect = yes
					scope:tog_mission_owner = {
						add_gold = minor_gold_value
						add_prestige = minor_prestige_gain
					}
				}
				# A success: the count converts.
				40 = {
					opinion_modifier = {
						who = scope:tog_missionary
						opinion_target = scope:tog_mission_owner
						multiplier = 0.6
					}
					save_scope_value_as = { name = tog_mission_result value = flag:won }
					tog_mission_convert_effect = yes
				}
				# A failure: the count does not listen.
				25 = {
					save_scope_value_as = { name = tog_mission_result value = flag:lost }
				}
				# Driven out: the count's men chase the missionary away.
				10 = {
					save_scope_value_as = { name = tog_mission_result value = flag:chased }
					scope:tog_mission_owner = { add_prestige = minor_prestige_loss }
					scope:tog_mission_target = {
						add_opinion = {
							target = scope:tog_mission_owner
							modifier = angry_opinion
							opinion = -20
						}
					}
				}
				# Wounded: the priest's men beat the missionary.
				5 = {
					save_scope_value_as = { name = tog_mission_result value = flag:wounded }
					scope:tog_missionary = {
						increase_wounds_no_death_effect = { REASON = fight }
					}
				}
			}
			# For the recap in the quest window.
			if = {
				limit = { scope:tog_story.var:tog_eogan ?= scope:tog_missionary }
				scope:tog_story = {
					set_variable = {
						name = tog_res_tog_eogan
						value = scope:tog_mission_result
					}
				}
			}
			if = {
				limit = { scope:tog_story.var:tog_nest ?= scope:tog_missionary }
				scope:tog_story = {
					set_variable = {
						name = tog_res_tog_nest
						value = scope:tog_mission_result
					}
				}
			}
			if = {
				limit = { scope:tog_story.var:tog_gofraid ?= scope:tog_missionary }
				scope:tog_story = {
					set_variable = {
						name = tog_res_tog_gofraid
						value = scope:tog_mission_result
					}
				}
			}
			if = {
				limit = { scope:tog_story.var:tog_ursin ?= scope:tog_missionary }
				scope:tog_story = {
					set_variable = {
						name = tog_res_tog_ursin
						value = scope:tog_mission_result
					}
				}
			}
			if = {
				limit = { scope:tog_story.var:tog_gwion ?= scope:tog_missionary }
				scope:tog_story = {
					set_variable = {
						name = tog_res_tog_gwion
						value = scope:tog_mission_result
					}
				}
			}
			if = {
				limit = { scope:tog_story.var:tog_peredur ?= scope:tog_missionary }
				scope:tog_story = {
					set_variable = {
						name = tog_res_tog_peredur
						value = scope:tog_mission_result
					}
				}
			}
			if = {
				limit = { scope:tog_story.var:tog_lailoken ?= scope:tog_missionary }
				scope:tog_story = {
					set_variable = {
						name = tog_res_tog_lailoken
						value = scope:tog_mission_result
					}
				}
			}
			scope:tog_mission_owner = { trigger_event = tog_quest.0712 }
		}
		remove_variable = tog_mission_target
		remove_variable = tog_mission_owner
		scope:tog_mission_owner ?= { tog_mission_result_in_effect = yes }
	}
