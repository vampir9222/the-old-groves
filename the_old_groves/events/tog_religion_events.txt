namespace = tog_religion

# The Archdruidate takes its arms and its name (see tog_on_title_gain_archdruid).
tog_religion.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		religion = religion:tog_celtic_religion
		exists = faith.religious_head_title
	}

	immediate = {
		faith.religious_head_title = {
			if = {
				limit = { NOT = { has_variable = tog_archdruid_coa } }
				set_coa = tog_archdruidate
				# "Archdruidate" rather than the vanilla "Druidic Archdruidate".
				set_title_name = tog_religious_head_title_name
				set_variable = { name = tog_archdruid_coa value = yes }
			}
		}
	}
}

# Lia Fáil: the stone of Tara cries out under the rightful king, or stays silent
# (see tog_on_title_gain_lia_fail). Odds in tog_lia_fail_cry_chance.
tog_religion.0010 = {
	type = character_event
	title = tog_religion.0010.t
	desc = {
		desc = tog_religion.0010.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:tog_lia_fail_rightful }
				desc = tog_religion.0010.desc.rightful
			}
			desc = tog_religion.0010.desc.usurper
		}
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:tog_lia_fail_cries }
				desc = tog_religion.0010.desc.cries
			}
			desc = tog_religion.0010.desc.silent
		}
	}
	theme = faith
	override_background = { reference = fp1_runestone }
	center_portrait = {
		character = root
		triggered_animation = {
			trigger = { exists = scope:tog_lia_fail_cries }
			animation = happiness
		}
		animation = worry
	}

	immediate = {
		random = {
			chance = tog_lia_fail_cry_chance
			save_scope_value_as = { name = tog_lia_fail_cries value = yes }
		}
	}

	option = {
		name = tog_religion.0010.a
		trigger = { exists = scope:tog_lia_fail_cries }
		add_legitimacy = minor_legitimacy_gain
		add_prestige = 100
	}

	option = {
		name = tog_religion.0010.b
		trigger = { NOT = { exists = scope:tog_lia_fail_cries } }
		add_legitimacy = miniscule_legitimacy_loss
	}
}
